
import 'package:flutter/material.dart';
void main(){
  runApp(MyApp);

}
class MyApp extends StatelessWidget{
  MyApp({super.key});
  Widget build(BuildeContext context){
    return MaterialApp(
      home:antima(),
     debugShowCheckedModeBanner: false,
    );
  }
}
class antima extends StatefulWidget{
  antima({super.key});
  State<antima> createState()=>_antimaState();
 
}
class _antimaState extends State<antima>{
  bool isGreen =false;
  void change(){
    setState((){
      isGreen=!isGreen;

    });
  }
 Widget build(BuildContext context){
  return Scaffold(
    appBar:AppBar(
      title:Text(
        'Traffic Signal Animation',
      centerTitle:true,
      ),
      body:Center(
        child:Column(
          children:[
            AntimatedContainer(
              duration:Duration(seconds:1),
              width:isGreen?200:150,
              height:isGreen?200:150,
              colors:isGreen?Colors.green:Colors.red,
            ),
            ElevatedButton(
              onPressed:change,
              child:Text('Change Signal'),
            ),
          ],
        ),
      ),
  );
 }
}
