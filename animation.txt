
import 'package:flutter/material.dart';

void main() {
  runApp(MaterialApp(
    home: TrafficSignal(),
    debugShowCheckedModeBanner: false,
  ));
}

class TrafficSignal extends StatefulWidget {
  @override
  State<TrafficSignal> createState() => _TrafficSignalState();
}

class _TrafficSignalState extends State<TrafficSignal> {

  // Initial values
  bool isGreen = false;

  // Change Signal
  void changeSignal() {
    setState(() {
      isGreen = !isGreen;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Traffic Signal Animation'),
      ),

      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [

            // Animated Container
            AnimatedContainer(
              duration: Duration(seconds: 1),
              width: isGreen ? 200 : 150,
              height: isGreen ? 200 : 150,
              color: isGreen ? Colors.green : Colors.red,
            ),

            SizedBox(height: 30),

            // Button
            ElevatedButton(
              onPressed: changeSignal,
              child: Text('Change Signal'),
            ),

          ],
        ),
      ),
    );
  }
}
