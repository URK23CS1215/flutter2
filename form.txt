
import 'package:flutter/material.dart';

void main() {
  runApp(MaterialApp(
    home: LibraryRegistration(),
    debugShowCheckedModeBanner: false,
  ));
}

class LibraryRegistration extends StatefulWidget {
  @override
  State<LibraryRegistration> createState() =>
      _LibraryRegistrationState();
}

class _LibraryRegistrationState extends State<LibraryRegistration> {

  // Create Form Key
  final _formKey = GlobalKey<FormState>();

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Library Membership Registration'),
      ),

      body: Padding(
        padding: EdgeInsets.all(20),

        child: Form(
          key: _formKey,

          child: Column(
            children: [

              // Student Name
              TextFormField(
                decoration: InputDecoration(
                  labelText: 'Student Name',
                  border: OutlineInputBorder(),
                ),
                validator: (value) {
                  if (value == null || value.isEmpty) {
                    return 'Enter Student Name';
                  }
                  return null;
                },
              ),

              SizedBox(height: 15),

              // Register Number
              TextFormField(
                decoration: InputDecoration(
                  labelText: 'Register Number',
                  border: OutlineInputBorder(),
                ),
                validator: (value) {
                  if (value == null || value.isEmpty) {
                    return 'Enter Register Number';
                  }
                  return null;
                },
              ),

              SizedBox(height: 15),

              // Email Address
              TextFormField(
                decoration: InputDecoration(
                  labelText: 'Email Address',
                  border: OutlineInputBorder(),
                ),
                validator: (value) {
                  if (value == null || value.isEmpty) {
                    return 'Enter Email Address';
                  }
                  return null;
                },
              ),

              SizedBox(height: 20),

              // Register Button
              ElevatedButton(
                onPressed: () {
                  if (_formKey.currentState!.validate()) {
                    ScaffoldMessenger.of(context).showSnackBar(
                      SnackBar(
                        content: Text(
                          'Membership Registered Successfully',
                        ),
                      ),
                    );
                  }
                },
                child: Text('Register'),
              ),

            ],
          ),
        ),
      ),
    );
  }
}
