
import 'dart:convert';
import 'package:flutter/material.dart';

// Employee Class
class Employee {
  int id;
  String name;
  String department;

  Employee(this.id, this.name, this.department);

  // Convert Object into JSON Map
  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'name': name,
      'department': department
    };
  }

  // Convert JSON Map into Object
  factory Employee.fromJson(Map<String, dynamic> json) {
    return Employee(
      json['id'],
      json['name'],
      json['department']
    );
  }
}

void main() {
  runApp(MaterialApp(
    home: EmployeePage(),
    debugShowCheckedModeBanner: false,
  ));
}

// StatefulWidget
class EmployeePage extends StatefulWidget {
  @override
  State<EmployeePage> createState() => _EmployeePageState();
}

class _EmployeePageState extends State<EmployeePage> {

  // Create Employee Object
  Employee emp = Employee(201, 'Ravi', 'IT');

  // Employee Name List
  List<String> employees = [];

  // Add Employee
  void addEmployee() {
    setState(() {
      employees.add(emp.name);
    });
  }

  @override
  Widget build(BuildContext context) {

    // Convert Employee into JSON
    String jsonData = jsonEncode(emp.toJson());

    return Scaffold(
      appBar: AppBar(
        title: Text('Employee Record Manager'),
      ),

      body: Padding(
        padding: EdgeInsets.all(20),

        child: Column(
          children: [

            Text(
              'Employee Details: ID ${emp.id} | '
              'Name ${emp.name} | '
              'Department ${emp.department}',
            ),

            SizedBox(height: 20),

            ElevatedButton(
              onPressed: addEmployee,
              child: Text('Add Employee'),
            ),

            SizedBox(height: 20),

            Text('Employee List:'),

            // Display Employee List
            Expanded(
              child: ListView.builder(
                itemCount: employees.length,
                itemBuilder: (context, index) {
                  return ListTile(
                    title: Text(employees[index]),
                  );
                },
              ),
            ),

            // Display JSON
            Text('JSON: $jsonData'),
          ],
        ),
      ),
    );
  }
}
